-- 配件 / Parts 库存 → Output-NZ/parts.csv
-- 复制为 parts.txt 后按业务修改；parts.txt 不提交 Git（每人本地 SQL）
-- 跨仓借调：无母件列时按 SKU 前两段或 ProductFamily 归组；也可放 Data-NZ/parts_kits.csv 定义母件/子件

SELECT
    w.Name AS WarehouseName,
    p.Sku,
    p.Name AS ProductName,
    ISNULL(p.ProductFamily, '') AS ProductFamily,
    SUM(s.Quantity) AS PartsQty
FROM [dbo].[Stocks] s
INNER JOIN [dbo].[Warehouses] w
    ON s.WarehouseId = w.Id
INNER JOIN [dbo].[Products] p
    ON s.ProductId = p.Id
WHERE s.StockStatus = 'Normal'
  AND s.StockOnHoldStatus IS NULL
  AND (
        p.ProductFamily LIKE '%Part%'
     OR p.ProductFamily LIKE '%配件%'
     OR p.Sku LIKE '%-PART%'
  )
GROUP BY
    w.Name,
    p.Sku,
    p.Name,
    p.ProductFamily
HAVING SUM(s.Quantity) > 0
ORDER BY w.Name, PartsQty DESC, p.Sku;
