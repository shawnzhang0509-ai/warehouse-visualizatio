-- On Hold / 冻结库存 → Output-NZ/on_hold.csv
-- 复制为 on_hold.txt 后按业务修改；on_hold.txt 不提交 Git（每人本地 SQL）
--
-- 列名看板会自动识别：WarehouseName、Sku、OnHoldQty、StockOnHoldStatus 等
-- 可选冻结日期列（用于「On Hold 分析」算天数）：OnHoldDate / HoldSince / ModifiedOn 等

SELECT
    w.Name AS WarehouseName,
    p.Sku,
    p.Name AS ProductName,
    ISNULL(p.ProductFamily, '') AS ProductFamily,
    s.StockOnHoldStatus,
    -- MIN(s.OnHoldDate) AS OnHoldDate,  -- 有该字段时取消注释
    SUM(s.Quantity) AS OnHoldQty
FROM [dbo].[Stocks] s
INNER JOIN [dbo].[Warehouses] w
    ON s.WarehouseId = w.Id
INNER JOIN [dbo].[Products] p
    ON s.ProductId = p.Id
WHERE s.StockStatus = 'Normal'
  AND s.StockOnHoldStatus IS NOT NULL
  AND s.Quantity > 0
GROUP BY
    w.Name,
    p.Sku,
    p.Name,
    p.ProductFamily,
    s.StockOnHoldStatus
ORDER BY w.Name, OnHoldQty DESC, p.Sku;
